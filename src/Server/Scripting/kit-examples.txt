@@ Start | Values kept between sessions | kept
@ top
-- kept.anything = value in any function: saved with the game, back after a reload.

@@ On the face | A number with − and + | kept
@ top
local {$shown}
@ draw
kept.{$count} = kept.{$count} or 0
{$shown} = self.button{ x = 0, z = {bottom}, label = tostring(kept.{$count}), font = {big} }
self.button{ x = {left}, z = {bottom}, w = {btn}, h = {btn}, label = "−", font = {font}, on_click = "{$less}" }
self.button{ x = {right}, z = {bottom}, w = {btn}, h = {btn}, label = "+", font = {font}, on_click = "{$more}" }
@ functions
function {$less}(who)
  kept.{$count} = kept.{$count} - 1
  self.edit({$shown}, { label = tostring(kept.{$count}) })
end

function {$more}(who)
  kept.{$count} = kept.{$count} + 1
  self.edit({$shown}, { label = tostring(kept.{$count}) })
end

@@ Right-click menu | Report its name and counters |
@ draw
self.menu_add("Report", "{$report}")
@ functions
function {$report}(who)
  local parts = {}
  for name, value in pairs(self.counters) do parts[#parts + 1] = name .. " " .. value end
  table.sort(parts)
  local text = self.name or self.kind_name or "Piece"
  if #parts > 0 then text = text .. ": " .. table.concat(parts, ", ") end
  game.say(text)
end

@@ Right-click menu | Distance to the nearest piece |
@ draw
self.menu_add("Nearest piece", "{$nearest}")
@ functions
function {$nearest}(who)
  local best, far
  for _, p in ipairs(board.pieces()) do
    if p.id ~= self.id then
      local d = board.distance(self, p)
      if d and (not far or d < far) then best, far = p, d end
    end
  end
  if best then
    game.say((best.name or best.kind_name or best.id) .. ": " .. math.floor(far + 0.5), { to = who })
  end
end

@@ Right-click menu | Range ring on and off | kept
@ draw
self.menu_add("Range ring", "{$ring}")
if kept.{$ringed} then {$draw}() end
@ functions
function {$ring}(who)
  kept.{$ringed} = not kept.{$ringed}
  if kept.{$ringed} then {$draw}() else self.set_lines({}) end
end

function {$draw}()
  local points = {}
  for i = 0, 71 do
    local a = i / 72 * math.pi * 2
    points[#points + 1] = { x = math.cos(a) * {ring}, z = math.sin(a) * {ring} }
  end
  self.set_lines{ { points = points, color = "#C0392B", width = 0.08, loop = true } }
end

@@ Right-click menu | Next state, round and round | states
@ draw
self.menu_add("Next state", "{$next}")
@ functions
function {$next}(who)
  self.set_state(self.state % #self.states + 1)
end

@@ When something happens | Say when it changes state |
@ on_state(old, new)
game.say((self.name or self.kind_name or "Piece") .. " is now " .. (self.states[new] or tostring(new)))

@@ When something happens | Say when it is turned over |
@ on_flip(face_down)
game.say((self.name or self.kind_name or "Piece") .. (face_down and " is face down" or " is face up"))

@@ When something happens | Say how far it moved | kept
@ on_pick_up(who)
kept.{$from} = { x = self.pos.x, z = self.pos.z }
@ on_drop(who)
if kept.{$from} then
  local d = board.distance(kept.{$from}, self.pos)
  if d then game.say((self.name or self.kind_name or "Piece") .. " moved " .. math.floor(d + 0.5), { to = who }) end
  kept.{$from} = nil
end

@@ When something happens | Count up each turn | counter
@ on_turn(turn)
self.set_counter("{counter}", self.counters["{counter}"] + 1)

@@ When something happens | A typed number sets a counter | counter
@ on_number_typed(who, n)
self.set_counter("{counter}", n)
return true
